TIME_INTEGRATORS = ['split_explicit', 'rk4']

class RealisticGlobalForward(Task):
    """
    A short forward run of the configured ocean model for a realistic global
    ocean simulation on one MPAS mesh.

    The task runs a single
    :py:class:`~polaris.tasks.ocean.realistic_global.forward.forward.Forward`
    step (``short``, a brief smoke test) whose duration and cadence come from
    the ``[realistic_global_forward]`` config section, with any per-mesh
    overrides applied on top.  Each mesh has a ``split_explicit`` variant of
    the task, which uses the default split time stepping, and an ``rk4``
    variant, which adds ``rk4.cfg`` on top of the per-mesh options to run
    RK4 in both models.  The target model is resolved from ``[ocean] model``
    during component setup.  Two diagnostic steps that are
    not run by default follow it: ``global_stats`` plots time series of the
    run's global statistics, and ``viz`` plots global maps of its state.

    Where the model's input files come from is the initial condition's
    business, not the task's, which is what lets one task cover two rather
    different costs.  A ``StepInitialCondition`` consumes the outputs of the
    shared ``realistic_global/init`` steps, which the caller passes as
    ``init_steps`` so the task is runnable end to end; a
    ``DatabaseInitialCondition`` downloads a cached initial condition and needs
    no upstream steps at all.

    Attributes
    ----------
    mesh_name : str
        The name of the MPAS mesh.

    time_integrator : str
        The variant of the task, ``'split_explicit'`` or ``'rk4'``.
    """

    def __init__(
        self,
        component,
        mesh_name: str,
        init_condition: InitialCondition,
        subdir_name: str = 'forward',
        init_steps: Optional[Dict[str, 'Step']] = None,
        time_integrator: str = 'split_explicit',
    ):
        """
        Create the task.

        Parameters
        ----------
        component : polaris.tasks.ocean.Ocean
            The ocean component the task belongs to.

        mesh_name : str
            The name of the MPAS mesh (e.g. ``'icos240km'``).

        init_condition : InitialCondition
            The source of the model input files.

        subdir_name : str, optional
            The name of the task's directory under the mesh.  It also keeps
            two tasks on the same mesh apart when they differ only in where
            their initial condition comes from.

        init_steps : dict of polaris.Step, optional
            Shared upstream steps to add to the task, keyed by the symlink name
            to give them.  Needed when ``init_condition`` consumes their
            outputs, and left unset for a source that does not.

        time_integrator : str, optional
            The variant of the task, one of ``TIME_INTEGRATORS``: the default
            split time stepping or RK4 in both models.  It is also the name
            of the task's directory under ``subdir_name``.
        """
        if time_integrator not in TIME_INTEGRATORS:
            raise ValueError(
                f'Unexpected time_integrator {time_integrator!r}; expected '
                f'one of {TIME_INTEGRATORS}'
            )
        base = (
            f'spherical/realistic_global/{mesh_name}/{subdir_name}/'
            f'{time_integrator}'
        )
        super().__init__(
            component=component,
            name='realistic_global_forward',
            subdir=base,
        )
        self.mesh_name = mesh_name
        self.time_integrator = time_integrator

        config_filename = 'forward.cfg'
        filepath = f'{component.name}/{base}/{config_filename}'
        config = PolarisConfigParser(filepath=filepath)
        # the colormaps of the viz step
        config.add_from_package(
            'polaris.tasks.ocean.realistic_global', 'realistic_global.cfg'
        )
        config.add_from_package(
            'polaris.tasks.ocean.realistic_global.forward', config_filename
        )
        add_realistic_global_mesh_config(config=config, mesh_name=mesh_name)
        if time_integrator == 'rk4':
            config.add_from_package(
                'polaris.tasks.ocean.realistic_global.forward', 'rk4.cfg'
            )
        self.set_shared_config(config, link=config_filename)

        # the shared init steps carry their own init.cfg
        if init_steps is not None:
            for symlink, step in init_steps.items():
                self.add_step(step, symlink=symlink)

        forward_step = Forward(
            component=component,
            name='short',
            subdir=f'{base}/short',
            init_condition=init_condition,
            validate_class='state',
        )
        forward_step.set_shared_config(config, link=config_filename)
        self.add_step(forward_step)

        # both are diagnostics rather than part of the run, so neither is on by
        # default; the statistics time series is the more useful of the two on
        # the long spin-ups this task is a building block for
        stats_analysis = StatsAnalysis(
            component=component,
            indir=base,
            forward_step=forward_step,
        )
        stats_analysis.set_shared_config(config, link=config_filename)
        self.add_step(stats_analysis, run_by_default=False)

        viz = Viz(component=component, indir=base, forward=forward_step)
        viz.set_shared_config(config, link=config_filename)
        self.add_step(viz, run_by_default=False)
