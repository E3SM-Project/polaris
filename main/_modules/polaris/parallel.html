def check_mache_supports_placement():
    """
    Check that the installed mache can confine a launch to part of an
    allocation, and can do it a node at a time.

    A mache without placement takes none of the arguments Polaris now passes
    it, so a run against one fails partway through with a ``TypeError`` from
    deep inside the launcher rather than saying what is wrong.  Worse, a
    future mache that accepted the argument and ignored it would let a run
    appear to work while oversubscribing the machine -- no error, wrong
    results, and slower than running one step at a time.  Both are much
    better caught here.

    Raises
    ------
    RuntimeError
        If the installed mache has no placement support, or expresses a
        placement's cores as one set for the whole launch
    """
    parameters = inspect.signature(
        ParallelSystem.get_parallel_command
    ).parameters
    if 'placement' not in parameters:
        raise RuntimeError(
            f'The installed mache ({mache.__version__}) cannot confine a '
            f'launch to part of an allocation, which Polaris now requires.\n'
            f'Update mache to a release whose get_parallel_command() '
            f'takes a placement.'
        )

    _check_placement_cores_are_per_node()
    _check_placement_gpus_are_per_node()




def _check_placement_cores_are_per_node():
    """
    Check that a placement's cores are one set per node.

    An earlier mache took one flat set of unique cores for the whole launch,
    which cannot describe a launch spanning nodes on the machines whose
    launchers bind cores explicitly: core numbers are node-local, so two
    nodes both using core 0 is the ordinary case and a single set forbids
    it.

    This builds exactly that -- two nodes, each offering core 0 -- rather
    than reading a version, for the same reason the check above does: the
    capability is what matters and a version is only a proxy for it.
    """
    try:
        placement = ResourcePlacement(nodes=('a', 'b'), cores=((0,), (0,)))
        total_cores = placement.total_cores
    except (AttributeError, TypeError, ValueError) as exception:
        raise RuntimeError(
            f'The installed mache ({mache.__version__}) expresses a '
            f"placement's cores as one set for the whole launch, so Polaris "
            f'cannot place a step across nodes.\n'
            f'Update mache to a release that gives a placement one set '
            f'of cores per node.'
        ) from exception

    if total_cores != 2:
        raise RuntimeError(
            f'The installed mache ({mache.__version__}) accepted a placement '
            f'of one core on each of two nodes and reported {total_cores} '
            f'cores rather than 2, so Polaris cannot rely on what a '
            f'placement means.\n'
            f'Update mache to a release that gives a placement one set '
            f'of cores per node.'
        )


def _check_placement_gpus_are_per_node():
    """
    Check that a placement's GPUs are one set per node, and that mache can
    say what the machine calls them.

    An earlier mache took one flat list of unique GPU indices for the whole
    launch, so a GPU step spanning nodes on a machine where Polaris has to
    name its GPUs -- Aurora -- could not be placed at all.  The same mache
    gave every task of a placed launch on Aurora every GPU it was given,
    and could not translate an index into the tile name a task sees.
    """
    try:
        placement = ResourcePlacement(
            nodes=('a', 'b'), cores=((0,), (0,)), gpus=2, gpu_ids=((0,), (0,))
        )
    except (AttributeError, TypeError, ValueError) as exception:
        raise RuntimeError(
            f'The installed mache ({mache.__version__}) expresses a '
            f"placement's GPUs as one list for the whole launch, so Polaris "
            f'cannot place a GPU step across nodes.\n'
            f'Update mache to a release that gives a placement one set '
            f'of GPUs per node.'
        ) from exception

    if placement.gpu_ids != ((0,), (0,)) or not hasattr(
        ParallelSystem, 'get_gpu_device_names'
    ):
        raise RuntimeError(
            f'The installed mache ({mache.__version__}) does not give a '
            f'placement one set of GPUs per node, or cannot name a '
            f"machine's GPUs, so Polaris cannot rely on what a placement "
            f'means.\n'
            f'Update mache to a release that gives a placement one set '
            f'of GPUs per node.'
        )


@contextmanager
def launching_step_command():
    """
    Mark the framework's launch of a step's command, the only place
    :py:meth:`polaris.Component.run_parallel_command` may be called.

    A step launches at most one thing, and only through its command
    (``args``).  Under ``polaris parallel``, a launch from anywhere else is
    not confined to the step's part of the allocation, and a step with no
    command may be run on another node, where a launch would start from
    inside a job step.
    """
    token = _launching_step_command.set(True)
    try:
        yield
    finally:
        _launching_step_command.reset(token)


def is_launching_step_command():
    """
    Whether the framework is launching a step's command

    Returns
    -------
    launching : bool
        True inside :py:func:`polaris.parallel.launching_step_command`
    """
    return _launching_step_command.get()

def set_parallel_systems(tasks, config: PolarisConfigParser):
    """
    Set the active parallel system on every component referenced by the task
    and step graph.

    Parameters
    ----------
    tasks : dict of polaris.Task
        Tasks to scan for referenced components

    config : polaris.config.PolarisConfigParser
        The config to use in constructing the parallel systems
    """
    check_mache_supports_placement()

    for component in get_components_in_use(tasks):
        component.set_parallel_system(config)




def get_memory_per_node(parallel_system):
    """
    Get the memory a node has, in MB, or ``None`` if this machine has not
    said.

    Read from the attribute where a newer mache offers one and from the
    config option otherwise, because the option is arriving in mache while
    this is being written and Polaris should work either side of it.  A
    machine that says nothing leaves memory undeclared rather than guessed
    at: a wrong figure here would propagate into every step's default.
    """
    memory_per_node = getattr(parallel_system, 'memory_per_node', None)
    if memory_per_node:
        return int(memory_per_node)
    memory_per_node = parallel_system.get_config_int('memory_per_node')
    if memory_per_node:
        return int(memory_per_node)
    return None
